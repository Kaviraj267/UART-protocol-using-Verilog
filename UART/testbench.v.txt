//==============================================================================
// UART Testbench
// Connects transmitter to receiver and verifies operation
//==============================================================================

`timescale 1ns / 1ps

module uart_testbench;

    //==========================================================================
    // Parameters
    //==========================================================================
    parameter CLK_FREQ  = 50_000_000;   // 50 MHz
    parameter BAUD_RATE = 9600;         // 9600 baud
    parameter CLK_PERIOD = 20;          // 20ns = 50MHz
    
    //==========================================================================
    // Signals
    //==========================================================================
    reg         clk;
    reg         rst;
    
    // Transmitter signals
    reg  [7:0]  tx_data;
    reg         tx_start;
    wire        tx;
    wire        tx_busy;
    
    // Receiver signals
    wire [7:0]  rx_data;
    wire        rx_done;
    wire        rx_error;
    
    //==========================================================================
    // Instantiate UART Transmitter
    //==========================================================================
    uart_tx #(
        .CLK_FREQ(CLK_FREQ),
        .BAUD_RATE(BAUD_RATE)
    ) transmitter (
        .clk(clk),
        .rst(rst),
        .tx_data(tx_data),
        .tx_start(tx_start),
        .tx(tx),
        .tx_busy(tx_busy)
    );
    
    //==========================================================================
    // Instantiate UART Receiver
    //==========================================================================
    uart_rx #(
        .CLK_FREQ(CLK_FREQ),
        .BAUD_RATE(BAUD_RATE)
    ) receiver (
        .clk(clk),
        .rst(rst),
        .rx(tx),            // Connect TX to RX (loopback)
        .rx_data(rx_data),
        .rx_done(rx_done),
        .rx_error(rx_error)
    );
    
    //==========================================================================
    // Clock Generation (50 MHz = 20ns period)
    //==========================================================================
    initial begin
        clk = 0;
        forever #(CLK_PERIOD/2) clk = ~clk;
    end
    
    //==========================================================================
    // Test Stimulus
    //==========================================================================
    initial begin
        // Initialize
        rst      = 1;
        tx_data  = 8'd0;
        tx_start = 0;
        
        // Wait and release reset
        #100;
        rst = 0;
        #100;
        
        // Test 1: Send 'A' (0x41)
        $display("Time=%0t: Sending 0x41 ('A')", $time);
        tx_data  = 8'h41;
        tx_start = 1;
        #(CLK_PERIOD);
        tx_start = 0;
        
        // Wait for transmission to complete
        wait(tx_busy == 0);
        #1000;
        
        // Test 2: Send 'B' (0x42)
        $display("Time=%0t: Sending 0x42 ('B')", $time);
        tx_data  = 8'h42;
        tx_start = 1;
        #(CLK_PERIOD);
        tx_start = 0;
        
        wait(tx_busy == 0);
        #1000;
        
        // Test 3: Send '5' (0x35)
        $display("Time=%0t: Sending 0x35 ('5')", $time);
        tx_data  = 8'h35;
        tx_start = 1;
        #(CLK_PERIOD);
        tx_start = 0;
        
        wait(tx_busy == 0);
        #1000;
        
        // Test 4: Send 0xFF (all ones)
        $display("Time=%0t: Sending 0xFF", $time);
        tx_data  = 8'hFF;
        tx_start = 1;
        #(CLK_PERIOD);
        tx_start = 0;
        
        wait(tx_busy == 0);
        #1000;
        
        // Test 5: Send 0x00 (all zeros)
        $display("Time=%0t: Sending 0x00", $time);
        tx_data  = 8'h00;
        tx_start = 1;
        #(CLK_PERIOD);
        tx_start = 0;
        
        wait(tx_busy == 0);
        #1000;
        
        $display("Time=%0t: All tests completed!", $time);
        $finish;
    end
    
    //==========================================================================
    // Monitor Receiver Output
    //==========================================================================
    always @(posedge clk) begin
        if (rx_done) begin
            if (rx_error) begin
                $display("Time=%0t: ERROR! Frame error detected!", $time);
            end
            else begin
                $display("Time=%0t: Received data = 0x%02h ('%c')", 
                         $time, rx_data, rx_data);
            end
        end
    end
    
    //==========================================================================
    // Generate VCD file for waveform viewing
    //==========================================================================
    initial begin
        $dumpfile("uart_test.vcd");
        $dumpvars(0, uart_testbench);
    end

endmodule
